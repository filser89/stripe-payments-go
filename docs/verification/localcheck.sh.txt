#!/usr/bin/env bash
set -euo pipefail
cd /Users/serg/code/filser89/stripe-payments-go
export COMPOSE_PROJECT_NAME=feature2-delivery-check POSTGRES_USER=verification POSTGRES_PASSWORD=local-verification-only POSTGRES_DB=verification DB_PORT=15492 APP_PORT=18092 BASIC_AUTH_USERNAME=verification-user BASIC_AUTH_PASSWORD=verification-password STRIPE_SECRET_KEY=sk_test_local_verification APP_BASE_URL=http://localhost:18092 CHECKOUT_REQUEST_TIMEOUT=2s STRIPE_CALL_TIMEOUT=100ms STRIPE_RETRY_BUDGET=500ms STRIPE_MAX_ATTEMPTS=1 HTTP_READ_TIMEOUT=3s HTTP_WRITE_TIMEOUT=3s
 docker compose config --quiet
 curl --silent --output /dev/null --write-out 'health_before=%{http_code}\n' http://localhost:18092/healthz
 curl --silent --output /dev/null --write-out 'ready_before=%{http_code}\n' http://localhost:18092/readyz
 docker compose exec -T app /app/service probe

 curl --silent --output /dev/null --write-out 'ready_outage=%{http_code}\n' http://localhost:18092/readyz
 docker compose start db
 curl --silent --fail --retry 10 --retry-delay 1 --max-time 2 --output /dev/null --write-out 'ready_recovery=%{http_code}\n' http://localhost:18092/readyz
 cat > /private/tmp/feature2-migration-failure.yaml <<'YAML'
services:
  migrate:
    command: [migrate, invalid]
YAML
 if COMPOSE_FILE=compose.yaml:/private/tmp/feature2-migration-failure.yaml make up > /private/tmp/feature2-migration-failure.log 2>&1; then echo unexpected_migration_success; exit 1; fi
 docker inspect --format 'app_running_after_migration_failure={{.State.Running}}' feature2-delivery-check-app-1
 make up > /private/tmp/feature2-local-restore.log 2>&1
 docker compose exec -T app /app/service probe
 docker compose stop app
 docker inspect --format 'graceful_exit={{.State.ExitCode}}' feature2-delivery-check-app-1
