type buffer struct {
	mu   sync.Mutex
	data bytes.Buffer
}

func (b *buffer) Write(p []byte) (int, error) {
	b.mu.Lock()
	defer b.mu.Unlock()
	return b.data.Write(p)
}
func (b *buffer) take() []byte {
	b.mu.Lock()
	defer b.mu.Unlock()
	d := append([]byte(nil), b.data.Bytes()...)
	b.data.Reset()
	return d
}

type brokenReader struct{}

func (brokenReader) Read([]byte) (int, error) { return 0, errors.New("verification-read-fault") }
func (brokenReader) Close() error             { return nil }
func must(e error) {
	if e != nil {
		panic(e)
	}
}
func main() {
	ctx := context.Background()
	root := "/Users/serg/code/filser89/stripe-payments-go"
	url := "postgres://verification:local-verification-only@127.0.0.1:55492/verification?sslmode=disable"
	must(postgres.Migrate(ctx, url, os.DirFS(root+"/db/migrations"), "up"))
	control, e := pgxpool.New(ctx, url)
	must(e)
	defer control.Close()
	_, e = control.Exec(ctx, `DO $$ BEGIN IF NOT EXISTS (SELECT 1 FROM pg_roles WHERE rolname='verification_app') THEN CREATE ROLE verification_app LOGIN PASSWORD 'driver-only'; END IF; END $$; GRANT CONNECT ON DATABASE verification TO verification_app; GRANT USAGE ON SCHEMA public TO verification_app; GRANT SELECT,INSERT,UPDATE ON ALL TABLES IN SCHEMA public TO verification_app; GRANT USAGE,SELECT ON ALL SEQUENCES IN SCHEMA public TO verification_app`)
	must(e)
	appURL := "postgres://verification_app:driver-only@127.0.0.1:55492/verification?sslmode=disable"
	pool, e := pgxpool.New(ctx, appURL)
	must(e)
	defer pool.Close()
	logs := &buffer{}
	logger := slog.New(slog.NewJSONHandler(logs, nil))
	var mu sync.Mutex
	scenario := "accepted"
	objects := map[string]map[string]any{}
	var callback func(string, string)
	stub := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		must(r.ParseForm())
		mu.Lock()
		mode := scenario
		cb := callback
		mu.Unlock()
		if mode == "timeout" || mode == "canceled" {
			<-r.Context().Done()
			return
		}
		if mode == "rejected" {
			w.WriteHeader(400)
			_, _ = w.Write([]byte(`{"error":{"type":"invalid_request_error","code":"parameter_invalid_integer","message":"private-diagnostic"}}`))
			return
		}
		if mode == "server" {
			w.WriteHeader(500)
			_, _ = w.Write([]byte(`{"error":{"type":"api_error","message":"private-diagnostic"}}`))
			return
		}
		var obj map[string]any
		mu.Lock()
		if r.Method == "POST" {
			sid := "cs_test_" + uuid.NewString()
			obj = map[string]any{"id": sid, "object": "checkout.session", "client_reference_id": r.Form.Get("client_reference_id"), "metadata": map[string]string{"order_id": r.Form.Get("metadata[order_id]"), "operation_id": r.Form.Get("metadata[operation_id]")}, "amount_total": 2500, "currency": "usd", "mode": "payment", "livemode": false, "status": "open", "payment_status": "unpaid", "url": "https://checkout.stripe.com/c/pay/verification", "expires_at": time.Now().Add(23 * time.Hour).Unix(), "payment_intent": nil}
			objects[sid] = obj
		} else {
			obj = objects[strings.TrimPrefix(r.URL.Path, "/v1/checkout/sessions/")]
			if mode == "prepared" {
				obj["status"] = "expired"
				obj["url"] = ""
			}
		}
		if mode == "await_confirmation" {
			obj["status"] = "complete"
			obj["payment_status"] = "paid"
			obj["payment_intent"] = "pi_verification"
			obj["url"] = ""
		}
		if mode == "mismatch" {
			obj["client_reference_id"] = "mismatch"
		}
		mu.Unlock()
		if cb != nil {
			metadata := obj["metadata"].(map[string]string)
			cb(metadata["order_id"], metadata["operation_id"])
		}
		w.Header().Set("Content-Type", "application/json")
		w.Header().Set("Request-Id", "req_verification")
		must(json.NewEncoder(w).Encode(obj))
	}))
	defer stub.Close()
	repo := postgres.NewPaymentRepository(pool)
	gateway := stripeapi.New("sk_test_verification", stripeapi.Options{BackendURL: stub.URL, HTTPClient: stub.Client()})
	opts := payment.Options{Currency: "usd", MinAmount: 50, MaxAmount: 100000, Origin: "http://localhost:8080", SDKVersion: "v87.0.0", APIVersion: "2026-09-30.endive", RequestTimeout: 2 * time.Second, CallTimeout: 100 * time.Millisecond, RetryBudget: 500 * time.Millisecond, MaxAttempts: 1, Logger: logger}
	service := payment.New(repo, gateway, opts)
	env := map[string]string{"DATABASE_URL": url, "BASIC_AUTH_USERNAME": "verification-user", "BASIC_AUTH_PASSWORD": "verification-password", "STRIPE_SECRET_KEY": "sk_test_verification", "CHECKOUT_REQUEST_TIMEOUT": "2s", "STRIPE_CALL_TIMEOUT": "100ms", "STRIPE_RETRY_BUDGET": "500ms", "STRIPE_MAX_ATTEMPTS": "1", "HTTP_READ_TIMEOUT": "3s", "HTTP_WRITE_TIMEOUT": "3s"}
	cfg, e := config.Load(func(k string) string { return env[k] })
	must(e)
	must(cfg.ValidateServing())
	handler := web.New(cfg, logger, pool.Ping, web.NewCheckoutHandler(service, opts.RequestTimeout, logger)).Handler
	results := []map[string]any{}
	record := func(name string, status int, requestID string) {
		raw := logs.take()
		records := []map[string]any{}
		for _, line := range bytes.Split(raw, []byte("\n")) {
			if len(line) == 0 {
				continue
			}
			var item map[string]any
			must(json.Unmarshal(line, &item))
			records = append(records, item)
		}
		results = append(results, map[string]any{"scenario": name, "status": status, "request_id": requestID, "records": records})
	}
	request := func(path, body string, auth bool) *httptest.ResponseRecorder {
		r := httptest.NewRequest("POST", path, strings.NewReader(body))
		r.Header.Set("Content-Type", "application/json")
		if auth {
			r.SetBasicAuth("verification-user", "verification-password")
		}
		w := httptest.NewRecorder()
		handler.ServeHTTP(w, r)
		return w
	}
	body := func(key string) string {
		return fmt.Sprintf(`{"description":"Verification purchase","amount":2500,"request_key":%q}`, key)
	}
	key := uuid.NewString()
	w := request("/api/orders", body(key), true)
	record("accepted", w.Code, w.Header().Get("X-Request-ID"))
	var response struct {
		Order     struct{ ID string }
		Operation struct{ ID string }
	}
	must(json.Unmarshal(w.Body.Bytes(), &response))
	order := response.Order.ID
	w = request("/api/orders", body(key), true)
	record("established_replay", w.Code, w.Header().Get("X-Request-ID"))
	w = request("/api/orders", `{invalid`, true)
	record("input_rejection", w.Code, w.Header().Get("X-Request-ID"))
	w = request("/api/orders", body(uuid.NewString()), false)
	record("auth_rejection", w.Code, w.Header().Get("X-Request-ID"))
	for _, name := range []string{"unsupported_media", "oversized", "body_read"} {
		r := httptest.NewRequest("POST", "/api/orders", strings.NewReader(body(uuid.NewString())))
		r.SetBasicAuth("verification-user", "verification-password")
		r.Header.Set("Content-Type", "application/json")
		if name == "unsupported_media" {
			r.Header.Set("Content-Type", "text/plain")
		}
		if name == "oversized" {
			r.Body = http.NoBody
			r.Body = io.NopCloser(strings.NewReader(strings.Repeat("x", 4097)))
		}
		if name == "body_read" {
			r.Body = brokenReader{}
		}
		w := httptest.NewRecorder()
		handler.ServeHTTP(w, r)
		record(name, w.Code, w.Header().Get("X-Request-ID"))
	}
	for _, name := range []string{"timeout", "server", "rejected", "mismatch", "await_confirmation"} {
		mu.Lock()
		scenario = name
		mu.Unlock()
		w = request("/api/orders", body(uuid.NewString()), true)
		record(name, w.Code, w.Header().Get("X-Request-ID"))
	}
	mu.Lock()
	scenario = "prepared"
	mu.Unlock()
	w = request("/api/orders/"+order+"/checkout", fmt.Sprintf(`{"request_key":%q}`, uuid.NewString()), true)
	record("prepared_after_get_budget", w.Code, w.Header().Get("X-Request-ID"))
	fault := func(kind string) {
		_, e := control.Exec(ctx, fmt.Sprintf(`CREATE FUNCTION verification_fault() RETURNS trigger LANGUAGE plpgsql AS $$ BEGIN IF NEW.kind='%s' THEN RAISE EXCEPTION 'verification only'; END IF; RETURN NEW; END $$; CREATE CONSTRAINT TRIGGER verification_fault AFTER INSERT ON payment_history DEFERRABLE INITIALLY DEFERRED FOR EACH ROW EXECUTE FUNCTION verification_fault()`, kind))
		must(e)
	}
	clearFault := func() {
		_, e := control.Exec(ctx, `DROP TRIGGER verification_fault ON payment_history; DROP FUNCTION verification_fault()`)
		must(e)
	}
	mu.Lock()
	scenario = "accepted"
	mu.Unlock()
	fault("order_created")
	w = request("/api/orders", body(uuid.NewString()), true)
	record("database_acceptance", w.Code, w.Header().Get("X-Request-ID"))
	clearFault()
	fault("dispatch_started")
	w = request("/api/orders", body(uuid.NewString()), true)
	record("database_dispatch", w.Code, w.Header().Get("X-Request-ID"))
	clearFault()
	mu.Lock()
	callback = func(_, _ string) { fault("operation_state_changed") }
	mu.Unlock()
	w = request("/api/orders", body(uuid.NewString()), true)
	record("database_result", w.Code, w.Header().Get("X-Request-ID"))
	clearFault()
	mu.Lock()
	callback = func(_, _ string) {
		fault("operation_state_changed")
		_, e := control.Exec(ctx, `REVOKE SELECT ON payment_orders,payment_operations FROM verification_app`)
		must(e)
	}
	mu.Unlock()
	w = request("/api/orders", body(uuid.NewString()), true)
	record("database_result_and_read", w.Code, w.Header().Get("X-Request-ID"))
	_, e = control.Exec(ctx, `GRANT SELECT ON payment_orders,payment_operations TO verification_app`)
	must(e)
	clearFault()
	_, e = control.Exec(ctx, `REVOKE SELECT ON payment_orders FROM verification_app`)
	must(e)
	r := httptest.NewRequest("GET", "/api/orders/"+order, nil)
	r.SetBasicAuth("verification-user", "verification-password")
	readResponse := httptest.NewRecorder()
	handler.ServeHTTP(readResponse, r)
	record("database_read", readResponse.Code, readResponse.Header().Get("X-Request-ID"))
	_, e = control.Exec(ctx, `GRANT SELECT ON payment_orders TO verification_app`)
	must(e)
	mu.Lock()
	callback = func(order, op string) {
		_, e := control.Exec(ctx, `UPDATE payment_operations SET owner_token='verification-takeover',version=version+1 WHERE id=$1`, op)
		must(e)
	}
	mu.Unlock()
	w = request("/api/orders", body(uuid.NewString()), true)
	record("ownership_lost", w.Code, w.Header().Get("X-Request-ID"))
	mu.Lock()
	callback = nil
	scenario = "canceled"
	mu.Unlock()
	cancelCtx, cancel := context.WithCancel(ctx)
	go func() { time.Sleep(30 * time.Millisecond); cancel() }()
	_, _ = service.Create(cancelCtx, payment.CreateInput{Description: "Direct canceled purchase", Amount: 2500, RequestKey: uuid.NewString()})
	cancel()
	record("direct_canceled", 0, "")
	output, e := json.MarshalIndent(results, "", "  ")
	must(e)
	must(os.WriteFile("decoded-logs.json", output, 0600))
	fmt.Printf("retained %d scenarios in decoded-logs.json\n", len(results))
}
